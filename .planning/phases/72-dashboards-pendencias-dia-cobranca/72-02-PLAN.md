```php
// Phase 72 Plan 02 — SC#5: para surveys v15 (template_id != null), lê médias
// por dimensão via NpsScoreCalculator (respeita template_snapshot); para surveys
// legacy pré-v15, mantém os score_* legados persistidos em NpsResponse (dual-path
// preservado durante migração — Phase 73 remove os legados quando dashboards convertidos).
$calculator = app(\App\Services\Nps\NpsScoreCalculator::class);

$surveysPayload = $surveys->map(function ($s) use ($calculator) {
    if (!$s->response) {
        return null; // survey sem resposta — filtra abaixo
    }

    $isV15 = $s->template_id !== null;

    return [
        'id'                 => $s->id,
        'month_reference'    => $s->month_reference?->format('Y-m'),
        'completed_at'       => $s->completed_at?->toIso8601String(),
        'template_id'        => $s->template_id,
        'score_empresa'      => $isV15
            ? $calculator->compute($s->response, 'empresa')
            : $s->response->score_empresa,
        'score_analista'     => $isV15
            ? $calculator->compute($s->response, 'analista')
            : $s->response->score_analista,
        'score_estrategista' => $isV15
            ? $calculator->compute($s->response, 'estrategista')
            : $s->response->score_estrategista,
        'comment'            => $s->response->comment,
        'respondent_name'    => $s->response->respondent_name,
    ];
})->filter()->values()->all();
```

**Nota:** `$surveys` presume o eager-load de `response.answers` já feito antes desse bloco. Se não estiver, adicionar `->with('response.answers')` na query original de `$surveys` (verificar contexto do método).
