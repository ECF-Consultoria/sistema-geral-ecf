   ```php
   namespace App\Services;
   use Illuminate\Support\Facades\Http;

   class HubspotApiClient
   {
       private const BASE = 'https://api.hubapi.com';
       public function __construct(private ?string $token = null) {
           $this->token = $this->token ?? config('services.hubspot.access_token');
       }
       public function fetchDeal(string $id, array $properties): array {
           $res = Http::withToken($this->token)
               ->get(self::BASE . "/crm/v3/objects/deals/{$id}", ['properties' => implode(',', $properties)]);
           $res->throw();
           return $res->json();
       }
       public function fetchAssociatedCompanyId(string $dealId): ?string {
               $res = Http::withToken($this->token)
                   ->get(self::BASE . "/crm/v3/objects/deals/{$dealId}/associations/companies");
               if (!$res->ok()) return null;
               return $res->json('results.0.toObjectId');
       }
       public function fetchCompany(string $id, array $properties): array {
           $res = Http::withToken($this->token)
               ->get(self::BASE . "/crm/v3/objects/companies/{$id}", ['properties' => implode(',', $properties)]);
           $res->throw();
           return $res->json();
       }
   }
   ```

4. **`app/Http/Controllers/Api/HubspotWebhookController.php`** — receiver completo. Estrutura:
   ```php
   namespace App\Http\Controllers\Api;
   use App\Http\Controllers\Controller;
   use App\Models\Company;
   use App\Models\ContratoServico;
   use App\Models\HubspotEvento;
   use App\Models\Servico;
   use App\Services\HubspotApiClient;
   use Illuminate\Http\Request;
   use Illuminate\Support\Facades\DB;
   use Illuminate\Support\Facades\Log;

   class HubspotWebhookController extends Controller
   {
       public function receive(Request $request, HubspotApiClient $api)
       {
           $rawBody = $request->getContent();
           $secret = config('services.hubspot.client_secret');

           $sigHdr = $request->header('X-HubSpot-Signature-v3', '');
           $tsHdr  = $request->header('X-HubSpot-Request-Timestamp', '');

           // Validacao timestamp (replay < 5min)
           $ts = (int) $tsHdr;
           if ($ts < 1 || abs((time() * 1000) - $ts) > 5 * 60 * 1000) {
               $this->logInvalida($rawBody, 'timestamp invalido');
               return response()->json(['error' => 'unauthorized'], 401);
           }

           // Calcula expected: base64(hmac_sha256(secret, METHOD + URI + body + timestamp))
           $methodUriBody = $request->method() . $request->fullUrl() . $rawBody . $tsHdr;
           $expected = base64_encode(hash_hmac('sha256', $methodUriBody, $secret, true));

           if (!$sigHdr || !hash_equals($expected, $sigHdr)) {
               $this->logInvalida($rawBody, 'signature invalida');
               return response()->json(['error' => 'unauthorized'], 401);
           }

           // Decodifica payload — HubSpot manda array de eventos
           $events = json_decode($rawBody, true) ?? [];
           if (!is_array($events)) $events = [$events];

           foreach ($events as $event) {
               $evento = HubspotEvento::create([
                   'signature_valid' => true,
                   'portal_id' => $event['portalId'] ?? null,
                   'object_type' => $event['objectType'] ?? null,
                   'object_id' => (string)($event['objectId'] ?? ''),
                   'subscription_type' => $event['subscriptionType'] ?? null,
                   'property_name' => $event['propertyName'] ?? null,
                   'property_value' => $event['propertyValue'] ?? null,
                   'payload' => $event,
                   'status' => 'recebido',
               ]);
               $this->processar($evento, $api);
           }

           return response()->json(['ok' => true]);
       }

       private function processar(HubspotEvento $evento, HubspotApiClient $api): void
       {
           // Filtra
           if ($evento->subscription_type !== 'deal.propertyChange'
               || $evento->property_name !== 'dealstage'
               || $evento->property_value !== config('services.hubspot.stage_fechado_ganho_id')) {
               $evento->update(['status' => 'ignorado', 'processado_em' => now()]);
               return;
           }

           // Idempotencia
           $jaProcessado = HubspotEvento::where('object_id', $evento->object_id)
               ->whereNotNull('company_id_criada')
               ->exists();
           if ($jaProcessado) {
               $evento->update(['status' => 'ignorado', 'erro_msg' => 'Deal ja processado', 'processado_em' => now()]);
               return;
           }

           try {
               $dealProps = array_values(config('services.hubspot.props.deal'));
               $companyProps = array_values(config('services.hubspot.props.company'));

               $deal = $api->fetchDeal($evento->object_id, array_merge(['dealname','amount','dealstage'], $dealProps));
               $companyId = $api->fetchAssociatedCompanyId($evento->object_id);
               $hubCompany = $companyId ? $api->fetchCompany($companyId, $companyProps) : null;

               $company = $this->criarEmpresa($deal, $hubCompany);

               $evento->update(['status' => 'processado', 'company_id_criada' => $company->id, 'processado_em' => now()]);
               Log::channel('ecf-webhooks')->info('[HubSpot Webhook] Empresa criada', ['evento_id' => $evento->id, 'company_id' => $company->id]);
           } catch (\Throwable $e) {
               $evento->update(['status' => 'erro', 'erro_msg' => $e->getMessage(), 'processado_em' => now()]);
               Log::channel('ecf-webhooks')->error('[HubSpot Webhook] Falha processamento', ['evento_id' => $evento->id, 'msg' => $e->getMessage()]);
           }
       }

       private function criarEmpresa(array $deal, ?array $hubCompany): Company
       {
           $propsDeal = config('services.hubspot.props.deal');
           $propsCompany = config('services.hubspot.props.company');
           $dprops = $deal['properties'] ?? [];
           $cprops = $hubCompany['properties'] ?? [];

           return DB::transaction(function () use ($deal, $dprops, $cprops, $propsDeal, $propsCompany) {
               $company = Company::create([
                   'name' => $cprops[$propsCompany['name']] ?? $deal['properties']['dealname'] ?? 'Empresa HubSpot',
                   'cnpj' => $cprops[$propsCompany['cnpj']] ?? null,
                   'email_cliente' => $cprops[$propsCompany['email']] ?? null,
                   'telefone' => $cprops[$propsCompany['phone']] ?? null,
                   'nicho' => $dprops[$propsDeal['nicho']] ?? null,
                   'dor' => $dprops[$propsDeal['dor']] ?? null,
                   'vende_ml' => isset($dprops[$propsDeal['vende_ml']]) ? filter_var($dprops[$propsDeal['vende_ml']], FILTER_VALIDATE_BOOLEAN, FILTER_NULL_ON_FAILURE) : null,
                   'faturamento_mensal' => is_numeric($dprops[$propsDeal['faturamento_mensal']] ?? null) ? (float) $dprops[$propsDeal['faturamento_mensal']] : null,
                   'empresa_nova' => true,
                   'status' => 'pendente',
                   'active' => true,
               ]);

               // Tenta vincular serviço se nome bater
               $servicoNome = $dprops[$propsDeal['servico']] ?? null;
               $servico = $servicoNome ? Servico::where('nome', $servicoNome)->where('ativo', true)->first() : null;
               $valor = isset($dprops['amount']) ? (float) $dprops['amount'] : ($servico?->valor_padrao ?? 0);

               if ($servico) {
                   ContratoServico::create([
                       'company_id' => $company->id,
                       'servico_id' => $servico->id,
                       'valor_contratado' => $valor,
                       'data_contratacao' => now()->toDateString(),
                       'ativo' => true,
                   ]);
               } else {
                   $notes = collect([$company->notes, $servicoNome ? "Serviço (HubSpot): {$servicoNome}" : null])->filter()->join("\n");
                   $company->update(['notes' => $notes]);
               }

               return $company;
           });
       }

       private function logInvalida(string $rawBody, string $motivo): void
       {
           HubspotEvento::create([
               'signature_valid' => false,
               'payload' => ['raw' => substr($rawBody, 0, 65000), 'motivo' => $motivo],
               'status' => 'erro',
               'erro_msg' => $motivo,
           ]);
       }
   }
   ```
