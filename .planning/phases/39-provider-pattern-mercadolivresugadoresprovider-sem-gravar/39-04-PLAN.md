```php
/**
 * Phase 39: monta lookup `campaign_id => ['name', 'status']` via provider.
 * Substitui chamada direta ao AdmanService::fetchAllCampaigns (preservada
 * em loadCampaignsInfo legacy para callsites externos como CleanupSugadoresQuarentena).
 */
private function buildCampaignsInfoFromProvider(\App\Contracts\SugadoresAdsProvider $provider, Company $company): array
{
    try {
        $campaigns = $provider->fetchCampaigns($company);
    } catch (\Throwable $e) {
        Log::warning("[Sugadores] Não conseguiu listar campanhas via provider '{$provider->name()}' para empresa {$company->id}: " . $e->getMessage());
        return [];
    }
    $lookup = [];
    foreach ($campaigns as $c) {
        $id = (string) ($c['campaign_id'] ?? '');
        if ($id === '') continue;
        $lookup[$id] = [
            'name'   => $c['campaign_name']   ?? null,
            'status' => $c['campaign_status'] ?? null,
        ];
    }
    return $lookup;
}
```
