    ```php
    public function index(Request $request): Response
    {
        $user = $request->user();
        $aba = $request->query('aba', 'nao-lidas'); // default = não lidas

        $query = $user->notifications()->latest();

        if ($aba === 'nao-lidas') {
            $query->whereNull('read_at');
        } else {
            // aba "todas" — apenas últimos 30 dias (HIST-02)
            $query->where('created_at', '>=', now()->subDays(30));
        }

        $notificacoes = $query->paginate(15)->through(function ($n) {
            $autorId = $n->data['autor_user_id'] ?? null;
            return [
                'id'         => $n->id,
                'read_at'    => $n->read_at,
                'created_at' => $n->created_at,
                'data'       => $n->data,
                'autor_nome' => $autorId ? \App\Models\User::find($autorId)?->name : null,
            ];
        });

        return Inertia::render('Notificacoes/Index', [
            'notificacoes' => $notificacoes,
            'aba'          => $aba,
        ]);
    }
    ```
