```
namespace App\Http\Controllers;
use App\Models\BonusFaixa;
use App\Http\Requests\UpdateBonusFaixaRequest;
use Illuminate\Http\Request;
use Inertia\Inertia;

class DesempenhoConfigController extends Controller
{
    public function index() : Response  // Inertia::render('Desempenho/Configuracao', ['faixas' => ...])
    public function updateFaixa(UpdateBonusFaixaRequest $request, BonusFaixa $faixa)
    public function toggleActive(BonusFaixa $faixa, Request $request)  // sem body — apenas flip
}
```

`index()`:
- Retorna Inertia render `Desempenho/Configuracao` com props:
  - `faixas` = `BonusFaixa::orderBy('ordem')->get()` (INCLUI inativas — admin precisa ver+reativar).
- Sem query params.

`updateFaixa(UpdateBonusFaixaRequest, BonusFaixa)`:
- FormRequest valida payload (schema abaixo).
- `$faixa->update($request->validated())`.
- Retornar `back()->with('success', "Faixa \"{$faixa->nome}\" atualizada com sucesso.")` (pt-BR).

`toggleActive(BonusFaixa, Request)`:
- Verificar se toggle não deixa cobertura órfã — se a faixa está sendo desativada E ela é a única cobrindo alguma parte de [0,5], adicionar `Log::warning` "[Desempenho Config] Faixa {$faixa->slug} desativada — cobertura pode ficar incompleta" (não bloquear — admin sabe o que faz).
- `$faixa->update(['ativo' => !$faixa->ativo])`.
- Retornar `back()->with('success', $faixa->ativo ? "Faixa \"{$faixa->nome}\" reativada." : "Faixa \"{$faixa->nome}\" desativada.")`.

FormRequest `App\Http\Requests\UpdateBonusFaixaRequest`:
- `authorize()`: `return $this->user()?->isAdmin() === true;` (defesa em profundidade).
- `rules()`:
  ```
  return [
      'nome'      => ['required', 'string', 'max:100'],
      'descricao' => ['nullable', 'string', 'max:2000'],
      'nota_min'  => ['required', 'numeric', 'between:0,5'],
      'nota_max'  => ['required', 'numeric', 'between:0,5', 'gte:nota_min'],
      'ordem'     => ['required', 'integer', 'min:0'],
  ];
  ```
- `withValidator($validator)`:
  - Validar `nota_min < nota_max` (não pode ser igual EXCETO se a faixa alvo é a `maximo` [5.00, 5.00] — permitir igualdade quando o slug atual é `maximo`).
  - Validar sobreposição:
    ```
    $faixaAtual = $this->route('faixa');  // BonusFaixa via route model binding
    $novoMin = (float) $this->input('nota_min');
    $novoMax = (float) $this->input('nota_max');
    $outrasAtivas = BonusFaixa::where('id', '!=', $faixaAtual->id)->where('ativo', true)->get();
    foreach ($outrasAtivas as $outra) {
        // sobreposição: novo_min <= outra.nota_max E novo_max >= outra.nota_min
        if ($novoMin <= (float)$outra->nota_max && $novoMax >= (float)$outra->nota_min) {
            $validator->errors()->add(
                'nota_min',
                "Sobreposição com a faixa \"{$outra->nome}\" [{$outra->nota_min}, {$outra->nota_max}]."
            );
            break;
        }
    }
    ```
- Mensagens `messages()` em pt-BR para regras críticas: `'nota_max.gte' => 'Nota máxima deve ser maior ou igual à nota mínima.'`, `'nota_min.between' => 'Nota mínima deve estar entre 0 e 5.'`, `'nota_max.between' => 'Nota máxima deve estar entre 0 e 5.'`.
