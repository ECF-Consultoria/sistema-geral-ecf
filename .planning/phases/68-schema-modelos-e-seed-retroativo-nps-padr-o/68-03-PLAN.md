    **`up()`:**

    ```php
    public function up(): void
    {
        DB::transaction(function () {
            // 1. Criar (ou reutilizar) o template "NPS Padrão" — idempotente
            $padraoId = DB::table('nps_templates')->where('is_default', true)->value('id');

            if (! $padraoId) {
                $padraoId = DB::table('nps_templates')->insertGetId([
                    'nome'                    => 'NPS Padrão',
                    'descricao'               => 'Template legado — 3 perguntas fixas (estrategista, analista, empresa) escala 1-5. Fallback quando empresa não tem template específico.',
                    'active'                  => true,
                    'is_default'              => true,
                    'priority'                => 0,
                    'envio_automatico_mensal' => true,
                    'created_at'              => now(),
                    'updated_at'              => now(),
                ]);
            }

            // 2. Semear 3 perguntas fixas — idempotente por (template_id, dimensao)
            $perguntas = [
                [
                    'dimensao'     => 'estrategista',
                    'texto'        => 'O atendimento do {nome_estrategista}',
                    'obrigatoria' => true,
                    'ordem'        => 1,
                ],
                [
                    'dimensao'     => 'analista',
                    'texto'        => 'O atendimento do {nome_analista}',
                    // Não obrigatória — renderização é condicional em runtime (Phase 71 decide via company.analista_id)
                    'obrigatoria' => false,
                    'ordem'        => 2,
                ],
                [
                    'dimensao'     => 'empresa',
                    'texto'        => 'A ECF está atendendo suas expectativas?',
                    'obrigatoria' => true,
                    'ordem'        => 3,
                ],
            ];

            foreach ($perguntas as $spec) {
                $existente = DB::table('nps_template_questions')
                    ->where('template_id', $padraoId)
                    ->where('dimensao', $spec['dimensao'])
                    ->first();

                if ($existente) {
                    $questionId = $existente->id;
                } else {
                    $questionId = DB::table('nps_template_questions')->insertGetId([
                        'template_id'  => $padraoId,
                        'texto'        => $spec['texto'],
                        'tipo'         => 'escala',
                        'dimensao'     => $spec['dimensao'],
                        'obrigatoria'  => $spec['obrigatoria'],
                        'active'       => true,
                        'ordem'        => $spec['ordem'],
                        'created_at'   => now(),
                        'updated_at'   => now(),
                    ]);
                }

                // 3. Semear 5 options (escala 1..5) — idempotente por (question_id, peso)
                for ($i = 1; $i <= 5; $i++) {
                    $optExiste = DB::table('nps_template_options')
                        ->where('template_question_id', $questionId)
                        ->where('peso', $i)
                        ->exists();

                    if (! $optExiste) {
                        DB::table('nps_template_options')->insert([
                            'template_question_id' => $questionId,
                            'label'                => (string) $i,
                            'peso'                 => $i,
                            'ordem'                => $i,
                            'created_at'           => now(),
                            'updated_at'           => now(),
                        ]);
                    }
                }
            }

            // 4. Retro-associação de surveys legadas — WHERE template_id IS NULL
            // Rodar como UPDATE em massa (uma query) — atômico dentro da transaction.
            // Se algum row falhar por corrupção rara, catch individual + log warning + skip individual (spec ROADMAP).
            $afetados = 0;
            try {
                $afetados = DB::table('nps_surveys')
                    ->whereNull('template_id')
                    ->update(['template_id' => $padraoId, 'updated_at' => now()]);
            } catch (\Throwable $e) {
                // Fallback: iterar por survey pra isolar falha individual (rota defensiva)
                Log::warning('[Phase 68 Seed] UPDATE em massa falhou; caindo pra iteração por survey', [
                    'error' => $e->getMessage(),
                ]);
                DB::table('nps_surveys')->whereNull('template_id')->orderBy('id')->each(function ($survey) use ($padraoId, &$afetados) {
                    try {
                        DB::table('nps_surveys')->where('id', $survey->id)->update([
                            'template_id' => $padraoId,
                            'updated_at'  => now(),
                        ]);
                        $afetados++;
                    } catch (\Throwable $inner) {
                        Log::warning('[Phase 68 Seed] Falha ao retro-associar survey individual', [
                            'survey_id' => $survey->id,
                            'error'     => $inner->getMessage(),
                        ]);
                        // Não abort — próxima survey
                    }
                });
            }

            Log::info('[Phase 68 Seed] Template NPS Padrão semeado e retro-associação concluída', [
                'template_id'       => $padraoId,
                'surveys_afetadas'  => $afetados,
                'orphans_restantes' => DB::table('nps_surveys')->whereNull('template_id')->count(),
            ]);
        });
    }
    ```
