Debounce fetch (500ms):
```js
useEffect(() => {
  const t = setTimeout(async () => {
    setCarregandoAfetadas(true);
    // Não usa ziggy route helper aqui — passar url manual porque axios não tem CSRF (window.axios já configura)
    const r = await window.axios.get(route('nps.configuracao.templates.empresas-afetadas', template.id));
    setAfetadas(r.data);
    setCarregandoAfetadas(false);
  }, 500);
  return () => clearTimeout(t);
}, [selected, template.id]);
```

**Comment interno:** o endpoint retorna estado BASEADO NO PIVOT ATUAL (persistido), não no `selected` local. Portanto: mostrar o count depois de salvar OU exibir aviso "Salve para atualizar a estimativa". Trade-off: v15.1 pode permitir simulação de payload não-persistido; v15.0 fica com "salve para ver". Alternativa aceita: fetcher só dispara depois do salvar (não no useEffect de selected).

Adotar a **alternativa** — não fazer debounce sobre selected, e sim disparar fetch após `onSaved()` callback. Simpler + preciso.
</action>

```jsx
import AppLayout from '@/Layouts/AppLayout';
import { useState, useEffect } from 'react';
import { router } from '@inertiajs/react';
import { Button } from '@/Components/ui/button';
import { X } from 'lucide-react';
import { cn } from '@/lib/utils';

import TemplatesList         from '@/Components/Nps/Config/TemplatesList';
import TemplateEditForm      from '@/Components/Nps/Config/TemplateEditForm';
import QuestionEditor        from '@/Components/Nps/Config/QuestionEditor';
import OptionsEditor         from '@/Components/Nps/Config/OptionsEditor';
import ServiceScopesPicker   from '@/Components/Nps/Config/ServiceScopesPicker';
import PreviewFormulario     from '@/Components/Nps/Config/PreviewFormulario';

// Modo legado preservado — chamado via botão "Textos e perguntas extras (v13)"
// que abre modal com iframe pra rota /nps/configuracao/textos-legado (Phase 33).
// Alternativa: importar os componentes de textos aqui, mas isso duplica 400 LOC —
// modal com iframe é o caminho mais barato para preservar Phase 33 durante a
// migração v15.0.

export default function Configuracao({
    templates,
    tipos_pergunta,
    dimensoes_labels,
    servicos_disponiveis,
}) {
    const [selectedId, setSelectedId] = useState(templates[0]?.id ?? null);
    const [creating, setCreating]     = useState(false);
    const [previewData, setPreviewData] = useState(null);

    const selected = templates.find(t => t.id === selectedId);

    // Preview live debounced 300ms — dispara quando template selecionado muda ou draft state altera
    useEffect(() => {
        if (!selected) return;
        const timer = setTimeout(async () => {
            const draft = {
                nome:      selected.nome,
                descricao: selected.descricao,
                perguntas: (selected.questions ?? []).map(q => ({
                    texto:       q.texto,
                    tipo:        q.tipo,
                    dimensao:    q.dimensao,
                    obrigatoria: q.obrigatoria,
                    options:     (q.options ?? []).map(o => ({ label: o.label, peso: o.peso })),
                })),
            };
            try {
                const r = await window.axios.post(route('nps.configuracao.templates.preview'), draft);
                setPreviewData(r.data.template);
            } catch (e) { /* silencioso — preview é secundário */ }
        }, 300);
        return () => clearTimeout(timer);
    }, [selectedId, selected]);

    // Recarrega lista de templates quando um filho salva algo — só a prop 'templates'
    const refresh = () => router.reload({ only: ['templates'] });

    return (
        <AppLayout title="Configuração NPS">
            <div className="max-w-[1600px] mx-auto p-6">
                <header className="flex items-center justify-between mb-6">
                    <h1 className="text-2xl font-bold text-white">Templates de NPS</h1>
                    {/* Botão para acessar modo legado Phase 33 (não implementado neste plan — abre em nova aba) */}
                    <a href="/nps/configuracao/textos-legado" className="text-sm text-white/60 hover:text-white underline">
                        Textos e perguntas extras (legado v13)
                    </a>
                </header>

                <div className="grid grid-cols-1 lg:grid-cols-[320px_1fr] gap-6">
                    {/* Coluna esquerda — lista */}
                    <TemplatesList
                        templates={templates}
                        selectedId={selectedId}
                        onSelect={id => { setSelectedId(id); setCreating(false); }}
                        onCreate={() => { setCreating(true); setSelectedId(null); }}
                    />

                    {/* Coluna direita — editor + preview */}
                    <div className="grid grid-cols-1 xl:grid-cols-2 gap-6">
                        <div className="space-y-6">
                            <TemplateEditForm
                                template={creating ? null : selected}
                                onSaved={id => { setCreating(false); setSelectedId(id); refresh(); }}
                            />

                            {selected && !creating && (
                                <>
                                    <QuestionEditor
                                        template={selected}
                                        tipos={tipos_pergunta}
                                        dimensoesLabels={dimensoes_labels}
                                        onChange={refresh}
                                    />

                                    {(selected.questions ?? []).map(q => (
                                        <OptionsEditor
                                            key={q.id}
                                            template={selected}
                                            question={q}
                                            onChange={refresh}
                                        />
                                    ))}

                                    <ServiceScopesPicker
                                        template={selected}
                                        servicos={servicos_disponiveis}
                                        onSaved={refresh}
                                    />
                                </>
                            )}
                        </div>

                        {/* Preview live sticky à direita */}
                        <div className="hidden xl:block sticky top-6 self-start">
                            <div className="text-sm text-white/60 mb-2">Preview do formulário público</div>
                            {previewData
                                ? <PreviewFormulario template={previewData} mode="preview" />
                                : <div className="text-white/40 text-sm">Selecione um template para ver o preview.</div>}
                        </div>
                    </div>
                </div>
            </div>
        </AppLayout>
    );
}
```

**Trade-off documentado:** preview é regenerado a cada `selected` muda; se o admin edita uma pergunta, `refresh()` recarrega props (que atualiza `selected` do Inertia), e o useEffect regenera o preview automaticamente. Não precisa de state duplicado.

**Nota:** o link `/nps/configuracao/textos-legado` requer redirecionar a rota legada `nps.configuracao.index` (linha 105 do routes/web.php) para o novo controller quando essa URL for acessada. Ver Task T8.
