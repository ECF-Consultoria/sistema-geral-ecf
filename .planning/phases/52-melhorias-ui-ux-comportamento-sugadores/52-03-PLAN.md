       ```
       const [elapsed, setElapsed] = useState(0);
       ```
    3. Adicionar `useEffect` do cronômetro:
       ```
       useEffect(() => {
           if (!analyzing) return;
           // Tick de 1s até 30 → dispara reload + reseta
           const tick = setInterval(() => {
               setElapsed(prev => {
                   const next = prev + 1;
                   if (next >= 30) {
                       clearInterval(tick);
                       router.reload({ only: ['sugador'], preserveScroll: true });
                       setAnalyzing(false);
                       return 0;
                   }
                   return next;
               });
           }, 1000);
           return () => clearInterval(tick);
       }, [analyzing]);
       ```
    4. Handler `rodarAnalise`:
       ```
       function rodarAnalise() {
           if (!sugador?.company?.id) return;
           setAnalyzing(true);
           setElapsed(0);
           router.post(route('sugadores.analyze-company', sugador.company.id), {}, {
               preserveScroll: true,
               // NÃO seta setAnalyzing(false) no onFinish — o cronômetro controla o fim
               onError: () => { setAnalyzing(false); setElapsed(0); }, // rollback em falha
           });
       }
       ```
