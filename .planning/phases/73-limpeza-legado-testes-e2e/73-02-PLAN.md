```php
/**
 * Phase 73 Plan 02 — cálculo real de metric='nps' usando NpsScoreCalculator.
 *
 * Escopo: todas as NpsResponse completed no período da meta pertencentes às
 * empresas do escopo (goal->company_id OU goal->portfolio->companies).
 * Dual-path: surveys v15 (template_id != null) via NpsScoreCalculator::compute
 * dimension 'empresa'; surveys legacy usam score_empresa direto (não quebrar
 * histórico).
 *
 * Retorna: float 1..5 (média das notas empresa) ou null quando não há resposta
 * no período (semântica "sem dado" vs "zero pontos").
 */
private function computeNps(\App\Models\Goal $goal): ?float
{
    // Escopo de empresas da meta.
    $companyIds = $goal->company_id
        ? [$goal->company_id]
        : $goal->portfolio?->companies()->pluck('companies.id')->all() ?? [];

    if (empty($companyIds)) {
        return null;
    }

    // Período da meta — assumindo colunas period_start_date e period_end_date;
    // ADAPTAR ao schema real do Goal (verificar durante execução).
    $inicio = $goal->period_start_date ?? $goal->start_date ?? null;
    $fim    = $goal->period_end_date   ?? $goal->end_date   ?? null;

    if (! $inicio || ! $fim) {
        return null;
    }

    // Responses completed no período das empresas do escopo.
    $responses = \App\Models\NpsResponse::query()
        ->whereHas('survey', function ($q) use ($companyIds, $inicio, $fim) {
            $q->whereIn('company_id', $companyIds)
              ->where('status', 'completed')
              ->whereBetween('completed_at', [$inicio, $fim]);
        })
        ->with('survey', 'answers')
        ->get();

    if ($responses->isEmpty()) {
        return null;
    }

    $calculator = app(\App\Services\Nps\NpsScoreCalculator::class);
    $notas = $responses->map(function ($r) use ($calculator) {
        // v15: template_id populado — usa calculator; legacy: score_empresa direto
        if ($r->survey && $r->survey->template_id !== null) {
            return $calculator->compute($r, 'empresa');
        }
        return $r->score_empresa;
    })->filter(fn ($n) => $n !== null);

    if ($notas->isEmpty()) {
        return null;
    }

    return round((float) $notas->avg(), 2);
}
```
