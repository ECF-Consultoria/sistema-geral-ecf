    ```php
    private function submitResponseV15(\Illuminate\Http\Request $request, \App\Models\NpsSurvey $survey)
    {
        // Eager load do template + questions + options — fonte da validação dinamica.
        $survey->load('template.questions.options');

        if (!$survey->template) {
            // Consistência quebrada — não deveria acontecer, template_id não-null aponta para template.
            // Fallback defensivo: 422 com mensagem clara.
            return response()->json(['error' => 'Template do NPS nao encontrado.'], 422);
        }

        // ─── Constrói rules de validação a partir do template snapshot ──────
        $rules = [
            'respondent_name' => 'nullable|string|max:255',
            'comment'         => 'nullable|string|max:2000',
            'answers'         => 'array',
        ];

        $questionsById = [];
        $optionsByQuestion = [];

        foreach ($survey->template->questions as $q) {
            $questionsById[$q->id] = $q;
            $optionIds = $q->options->pluck('id')->all();
            $optionsByQuestion[$q->id] = $q->options->keyBy('id');

            $base = "answers.{$q->id}";
            $required = $q->obrigatoria ? 'required' : 'nullable';
            // Tipo integer + Rule::in(ids validos das options daquela pergunta).
            $rules[$base] = [$required, 'integer', \Illuminate\Validation\Rule::in($optionIds)];
        }

        $validated = $request->validate($rules);

        // ─── Persiste dentro de transacao — inclui update status='completed'.
        // QueryException 23000 pode ser disparado ao completar a survey se
        // dedup unique parcial (Plan 68-04) detectar duplicata do mes.
        try {
            \Illuminate\Support\Facades\DB::transaction(function () use ($survey, $validated, $questionsById, $optionsByQuestion) {

                // NpsResponse criado sem score_* — fonte v15.0 e nps_response_answers.
                $response = \App\Models\NpsResponse::create([
                    'survey_id'          => $survey->id,
                    'respondent_name'    => $validated['respondent_name'] ?? null,
                    'score_estrategista' => null,
                    'score_analista'     => null,
                    'score_empresa'      => null,
                    'comment'            => $validated['comment'] ?? null,
                ]);

                // 1 NpsResponseAnswer por pergunta respondida.
                foreach (($validated['answers'] ?? []) as $qid => $optionId) {
                    if ($optionId === null) {
                        continue; // pergunta opcional sem resposta
                    }
                    $question = $questionsById[$qid] ?? null;
                    $option   = $optionsByQuestion[$qid][$optionId] ?? null;
                    if (!$question || !$option) {
                        continue; // defesa contra ids tampered — validate ja pegou
                    }

                    \App\Models\NpsResponseAnswer::create([
                        'response_id'                 => $response->id,
                        'template_question_id'        => $question->id,
                        'template_option_id'          => $option->id,
                        'question_texto_snapshot'     => $question->texto,
                        'question_dimensao_snapshot'  => $question->dimensao,
                        'option_label_snapshot'       => $option->label,
                        'option_peso_snapshot'        => $option->peso,
                        'comentario'                  => null,
                    ]);
                }

                // Atualiza status=completed — DB dispara 23000 se duplicata.
                $survey->update([
                    'status'       => 'completed',
                    'completed_at' => now(),
                ]);
            });
        } catch (\Illuminate\Database\QueryException $e) {
            if ((string) $e->getCode() === '23000') {
                // Phase 68 Plan 68-04: dedup unique parcial (company_id, month_reference, template_id)
                // bloqueou 2a completacao no mesmo mes. UX: renderiza tela ja existente.
                return \Inertia\Inertia::render('Nps/AlreadyCompleted');
            }
            throw $e;
        }

        return \Inertia\Inertia::render('Nps/ThankYou');
    }
    ```
